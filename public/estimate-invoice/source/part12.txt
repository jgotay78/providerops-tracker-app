      function num(value) {
        const parsed = Number.parseFloat(value);
        return Number.isFinite(parsed) ? parsed : 0;
      }

      function positive(value) {
        return Math.max(0, num(value));
      }

      function currency(value) {
        const symbol = $("currencySymbol").value || "$";
        const number = Number.isFinite(value) ? value : 0;
        return `${number < 0 ? "−" : ""}${symbol}${Math.abs(number).toLocaleString(undefined, {
          minimumFractionDigits: 2,
          maximumFractionDigits: 2
        })}`;
      }

      function escapeHtml(value) {
        return String(value ?? "")
          .replaceAll("&", "&amp;")
          .replaceAll("<", "&lt;")
          .replaceAll(">", "&gt;")
          .replaceAll('"', "&quot;")
          .replaceAll("'", "&#039;");
      }


      function securePaymentUrl(value) {
        let raw = String(value || "").trim();
        if (!raw) return "";
        // Make common pasted payment links friendlier: paypal.me/name -> https://paypal.me/name
        if (!/^[a-zA-Z][a-zA-Z0-9+.-]*:/.test(raw)) raw = "https://" + raw.replace(/^\/+/, "");
        try {
          const url = new URL(raw);
          return url.protocol === "https:" && url.hostname ? url.href : "";
        } catch {
          return "";
        }
      }

      function paymentOptions() {
        const link = $("paymentLink").value.trim();
        const instructions = $("paymentInstructions").value.trim();
        const language = $("quoteLanguage").value;
        const output = [];
        if (securePaymentUrl(link)) output.push({key:"checkout", label: language === "es" ? "Pagar ahora" : language === "both" ? "Pay now / Pagar ahora" : "Pay now", value: link, url: securePaymentUrl(link)});
        if (instructions) output.push({key:"instructions",label:language === "es" ? "Instrucciones" : "Payment instructions",value:instructions,url:""});
        return output;
      }

      function primaryPaymentOption() {
        return paymentOptions().find(item => item.url) || null;
      }

      function paymentOptionsText(language) {
        const options = paymentOptions();
        if (!options.length || !$("includePaymentLink").checked) return "";
        const heading = language === "es" ? "Opciones de pago" : language === "both" ? "Payment options / Opciones de pago" : "Payment options";
        return [heading, ...options.map(item => `${item.label}: ${item.value}`)].join("\n");
      }

      function htmlToPlainText(value) {
        const node = document.createElement("div");
        node.innerHTML = value;
        return node.textContent || node.innerText || "";
      }

      async function copyPlainText(value, message) {
        const text = String(value || "");
        try {
          await navigator.clipboard.writeText(text);
        } catch {
          const textarea = document.createElement("textarea");
          textarea.value = text;
          textarea.style.position = "fixed";
          textarea.style.opacity = "0";
          document.body.appendChild(textarea);
          textarea.select();
          document.execCommand("copy");
          textarea.remove();
        }
        if (message) showToast(message);
      }

      async function copyRichHtml(htmlValue, plainValue) {
        try {
          if (window.ClipboardItem && navigator.clipboard && navigator.clipboard.write) {
            const item = new ClipboardItem({
              "text/html": new Blob([htmlValue], {type: "text/html"}),
              "text/plain": new Blob([plainValue], {type: "text/plain"})
            });
            await navigator.clipboard.write([item]);
            return true;
          }
        } catch {}
        try {
          const holder = document.createElement("div");
          holder.contentEditable = "true";
          holder.style.position = "fixed";
          holder.style.left = "-10000px";
          holder.innerHTML = htmlValue;
          document.body.appendChild(holder);
          const range = document.createRange();
          range.selectNodeContents(holder);
          const selection = window.getSelection();
          selection.removeAllRanges();
          selection.addRange(range);
          const copied = document.execCommand("copy");
          selection.removeAllRanges();
          holder.remove();
          return copied;
        } catch {
          await copyPlainText(plainValue);
          return false;
        }
      }

      function launchExternal(url) {
        if (!url) return;
        const opened = window.open(url, "_blank", "noopener");
        if (!opened) showToast("Your browser blocked the new tab. Your work is still here; allow pop-ups for this file and try again.");
      }

      }

      function isoToday() {
        const now = new Date();
        const local = new Date(now.getTime() - now.getTimezoneOffset() * 60000);
        return local.toISOString().slice(0, 10);
      }

      function addDays(dateString, days) {
        const base = dateString ? new Date(`${dateString}T12:00:00`) : new Date();
        base.setDate(base.getDate() + Math.max(1, Math.round(days || 14)));
        return base.toISOString().slice(0, 10);
      }

      function formatDate(dateString) {
        if (!dateString) return "";
        const date = new Date(`${dateString}T12:00:00`);
        return date.toLocaleDateString(state.uiLanguage === "es" ? "es-US" : "en-US", {
          year: "numeric",
          month: "short",
          day: "numeric"
        });
      }

      function makeQuoteNumber() {
        const prefix = ($("quotePrefix").value || "Q").trim().toUpperCase();
        const next = Math.max(1, Math.round(num($("nextQuoteNumber").value) || 1001));
        return `${prefix}-${next}`;
      }

      function showToast(message) {
        const toast = $("toast");
        toast.textContent = message;
        toast.classList.add("show");
        clearTimeout(toastTimer);
        toastTimer = setTimeout(() => toast.classList.remove("show"), 2600);
      }
