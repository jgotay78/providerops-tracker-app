      // Unlike New document, this clears *both* business and job fields. Only the
      // current draft is reset: previously saved estimates/invoices are retained.
      function clearAllFields() {
        if (!window.confirm("Clear ALL fields in the current form? Saved estimates and invoices will NOT be deleted.")) return;

        // Remove the persisted current draft first so nothing stale can repopulate
        // while this reset is in progress. Saved document history uses a separate key.
        try { localStorage.removeItem(STORAGE_KEY); } catch (_) {}

        // Clear every current-form field explicitly. Do not call New document here,
        // because that helper persists an intermediate state by design.
        const blankIds = [
          "businessName","businessPhone","businessEmail","businessAddress",
          "customerName","customerPhone","customerEmail","serviceAddress","serviceDate",
          "paymentLink","paymentInstructions","stripePaymentLink","paypalPayment",
          "venmoPayment","zellePayment","cashAppPayment","otherPaymentLabel","otherPayment",
          "notesEn","notesEs","termsEn","termsEs","internalNotes",
          "contactNotes","contactNext","contactSubject","contactMessage",
          "typedSignatureInput"
        ];
        blankIds.forEach(id => {
          const field = $(id);
          if (!field) return;
          field.value = "";
          // Help browsers forget their displayed/autofilled value immediately.
          field.removeAttribute("value");
        });

        const numericDefaults = {
          discount:"0", taxRate:"0", deposit:"0", validDays:"14",
          workers:"1", hoursEach:"0", laborRate:"0", materialsCost:"0",
          miles:"0", mileCost:"0", otherCost:"0", overheadCost:"0",
          processingRate:"0", targetMargin:"0", amountPaid:"0", newPaymentAmount:"0"
        };
        Object.entries(numericDefaults).forEach(([id,value]) => {
          const field = $(id); if (field) field.value = value;
        });

        // Reset document/payment controls to a predictable new-estimate state.
        if ($("logoInput")) $("logoInput").value = "";
        if ($("includePaymentLink")) $("includePaymentLink").checked = true;
        if ($("primaryPaymentMethod")) $("primaryPaymentMethod").value = "auto";
        if ($("quoteLanguage")) $("quoteLanguage").value = EDITION_LANGUAGE;
        if ($("documentType")) $("documentType").value = "estimate";
        if ($("invoiceStatus")) $("invoiceStatus").value = "unpaid";
        if ($("documentDate")) $("documentDate").value = isoToday();
        if ($("dueDate")) $("dueDate").value = addDays(isoToday(),14);
        if ($("contactSituation")) $("contactSituation").value = "inquiry";
        if ($("contactTone")) $("contactTone").value = "professional";

        // Reset the current document state while leaving saved history untouched.
        state.logoData = "";
        state.savedId = null;
        state.items = [];
        state.signatureData = "";
        state.signatureMode = "draw";
        state.typedSignature = "";
        state.signedDate = "";
        state.approvalKey = "";
        state.approvalRecord = null;
        state.lastPaymentAmount = 0;
        state.lastPaymentDate = "";
        state.contactNext = "";
        state.contactNotes = "";
        state.contactLog = [];
        state.contactDraft = {situation:"inquiry",tone:"professional",subject:"",message:""};

        // Start with exactly one blank service row.
        state.items.push({
          id: (crypto.randomUUID ? crypto.randomUUID() : `${Date.now()}-${Math.random()}`),
          descriptionEn:"", descriptionEs:"", qty:1, rate:0
        });

        // Keep document numbering/settings, but create a fresh current reference.
        state.quoteNumber = makeQuoteNumber();

        renderSignatureMode();
        renderItems();
        renderContactLog();
        renderContactQueue();
        calculateAndRender();

        // Persist exactly the blank draft. This makes the clear survive reopening.
        try { localStorage.setItem(STORAGE_KEY, JSON.stringify(captureForm())); } catch (_) {}
        showToast("All current fields cleared. Saved documents are unchanged.");
      }

      function exportBackup() {
        const data = {
          product: "Bilingual Service Quote, Invoice & Profit Builder",
          exportedAt: new Date().toISOString(),
          current: captureForm(),
          history: getHistory()
        };
        const blob = new Blob([JSON.stringify(data, null, 2)], {type: "application/json"});
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.download = `estimate-invoice-builder-backup-${isoToday()}.json`;
        document.body.appendChild(link);
        link.click();
        link.remove();
        URL.revokeObjectURL(url);
        showToast(i18n[state.uiLanguage].backupSaved);
      }

      async function importBackup(file) {
        try {
          const text = await file.text();
          const data = JSON.parse(text);
          const compatibleProducts = ["Bilingual Service Quote & Profit Builder", "Bilingual Estimate, Invoice & Profit Builder", "Bilingual Service Quote, Invoice & Profit Builder"];
          if (!compatibleProducts.includes(data.product) || !data.current || !Array.isArray(data.history)) throw new Error("Invalid");
          localStorage.setItem(HISTORY_KEY, JSON.stringify(data.history));
          restoreForm(data.current);
          persist();
          renderHistory(); renderContactQueue();
          showToast(i18n[state.uiLanguage].backupRestored);
        } catch {
          showToast(i18n[state.uiLanguage].invalidBackup);
        }
      }

      function handleLogo(file) {
        if (!file) return;
        if (file.size > 900000) {
          showToast(state.uiLanguage === "es" ? "Usa un logo menor de 900 KB." : "Please use a logo smaller than 900 KB.");
          $("logoInput").value = "";
          return;
        }
        const reader = new FileReader();
        reader.onload = () => {
          state.logoData = reader.result;
          calculateAndRender();
          persist();
        };
        reader.readAsDataURL(file);
      }

      }
