      function recordPaymentAndThank() {
        if (documentType() !== "invoice") {
          showToast(state.uiLanguage === "es" ? "Convierte primero el documento a factura." : "Convert the document to an invoice first.");
          return;
        }
        const calc = calculations();
        const previousPaid = Math.min(positive($("amountPaid").value), calc.total);
        const remainingBeforePayment = Math.max(0, calc.total - previousPaid);
        if (remainingBeforePayment <= 0.004) {
          showToast(i18n[state.uiLanguage].noBalanceDue);
          return;
        }
        const enteredPayment = positive($("newPaymentAmount").value);
        if (enteredPayment <= 0) {
          showToast(i18n[state.uiLanguage].invalidPaymentAmount);
          $("newPaymentAmount").focus();
          return;
        }
        if (enteredPayment > remainingBeforePayment + 0.004) {
          showToast(i18n[state.uiLanguage].paymentExceedsBalance);
          $("newPaymentAmount").focus();
          return;
        }
        const appliedPayment = enteredPayment;
        const newTotalPaid = previousPaid + appliedPayment;
        const remaining = Math.max(0, calc.total - newTotalPaid);
        $("amountPaid").value = newTotalPaid.toFixed(2);
        $("invoiceStatus").value = remaining <= 0.004 ? "paid" : "partial";
        state.lastPaymentAmount = appliedPayment;
        state.lastPaymentDate = isoToday();
        $("newPaymentAmount").value = "0";
        calculateAndRender();
        persist();
        saveQuote();
        showToast(i18n[state.uiLanguage].paymentRecorded);
        openPaymentThankYouModal();
      }

      function drawPaymentQr(text) {
        const wrap = $("paymentQrWrap");
        if (wrap.dataset.error === "1") wrap.innerHTML = '<canvas id="paymentQrCanvas" width="240" height="240"></canvas>';
        const canvas = $("paymentQrCanvas");
        const ctx = canvas.getContext("2d");
        try {
          const qr = new QRCore.QRCode(0, QRCore.ErrorLevel.M);
          qr.addData(text);
          qr.make();
          const n = qr.getModuleCount(), quiet = 4, moduleSize = Math.max(3, Math.floor(250 / (n + quiet * 2))), size = (n + quiet * 2) * moduleSize;
          canvas.width = size; canvas.height = size;
          ctx.fillStyle = "#fff"; ctx.fillRect(0, 0, size, size);
          ctx.fillStyle = "#000";
          for (let row = 0; row < n; row++) for (let col = 0; col < n; col++) if (qr.isDark(row, col)) ctx.fillRect((col + quiet) * moduleSize, (row + quiet) * moduleSize, moduleSize, moduleSize);
          canvas.hidden = false; wrap.dataset.error = "";
        } catch {
          canvas.hidden = true; wrap.dataset.error = "1";
          wrap.innerHTML = `<p>${escapeHtml(i18n[state.uiLanguage].noPaymentLink)}</p>`;
        }
      }
