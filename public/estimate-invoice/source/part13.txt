
      function translateUI(key, language = state.uiLanguage || "en") {
        return (i18n[language] && i18n[language][key]) || (i18n.en && i18n.en[key]) || "";
      }

      function syncContactSectionLanguage() {
        ["contactTitle","contactHelp","contactLog","contactQueue","contactDisclaimer"].forEach((key) => {
          const value = translateUI(key);
          document.querySelectorAll(`[data-i18n="${key}"]`).forEach((node) => {
            node.textContent = value;
          });
        });
      }

      function applyLanguage(language) {
        language=EDITION_LANGUAGE;
        $("quoteLanguage").value=EDITION_LANGUAGE;
        state.uiLanguage = language;
        document.documentElement.lang = language;
        $("uiEnglish").classList.toggle("active", language === "en");
        $("uiSpanish").classList.toggle("active", language === "es");
        document.querySelectorAll("[data-i18n]").forEach((node) => {
          const key = node.dataset.i18n;
          const value = translateUI(key, language);
          if (value) node.textContent = value;
        });
        syncContactSectionLanguage();
        applyContactLocalization(language);
        renderSignatureMode();
        renderItems();
        calculateAndRender();
        renderHistory();
        persist();
      }

      function addItem(descriptionEn = "", descriptionEs = "", qty = 1, rate = 0) {
        state.items.push({
          id: (crypto.randomUUID ? crypto.randomUUID() : `${Date.now()}-${Math.random()}`),
          descriptionEn,
          descriptionEs,
          qty,
          rate
        });
        renderItems();
        calculateAndRender();
      }

      function renderItems() {
        const body = $("itemsBody");
        body.innerHTML = "";
        state.items.forEach((item, index) => {
          const row = document.createElement("tr");
          row.innerHTML = `
            <td><input data-item-index="${index}" data-field="descriptionEn" value="${escapeHtml(item.descriptionEn)}" aria-label="English description"></td>
            <td><input data-item-index="${index}" data-field="descriptionEs" value="${escapeHtml(item.descriptionEs)}" aria-label="Spanish description"></td>
            <td><input class="num" type="number" min="0" step="0.01" data-item-index="${index}" data-field="qty" value="${positive(item.qty)}" aria-label="Quantity"></td>
            <td><input class="num" type="number" min="0" step="0.01" data-item-index="${index}" data-field="rate" value="${positive(item.rate)}" aria-label="Rate"></td>
            <td class="row-total">${currency(positive(item.qty) * positive(item.rate))}</td>
            <td><button class="icon-btn" type="button" data-remove-index="${index}" aria-label="Remove">×</button></td>
          `;
          body.appendChild(row);
        });

        body.querySelectorAll("input[data-item-index]").forEach((input) => {
          input.addEventListener("input", (event) => {
            const index = Number(event.target.dataset.itemIndex);
            const field = event.target.dataset.field;
            state.items[index][field] = ["qty","rate"].includes(field) ? positive(event.target.value) : event.target.value;
            const row = event.target.closest("tr");
            row.querySelector(".row-total").textContent = currency(positive(state.items[index].qty) * positive(state.items[index].rate));
            calculateAndRender();
            persist();
          });
        });

        body.querySelectorAll("[data-remove-index]").forEach((button) => {
          button.addEventListener("click", () => {
            state.items.splice(Number(button.dataset.removeIndex), 1);
            if (!state.items.length) addItem();
            renderItems();
            calculateAndRender();
            persist();
          });
        });
      }

      function calculations() {
        const subtotal = state.items.reduce((sum, item) => sum + positive(item.qty) * positive(item.rate), 0);
        const discount = Math.min(positive($("discount").value), subtotal);
        const serviceRevenue = Math.max(0, subtotal - discount);
        const taxRate = positive($("taxRate").value) / 100;
        const tax = serviceRevenue * taxRate;
        const total = serviceRevenue + tax;
        const deposit = Math.min(positive($("deposit").value), total);
        const balance = Math.max(0, total - deposit);

        const workerCount = positive($("workers").value);
        const hoursEach = positive($("hoursEach").value);
        const totalLaborHours = workerCount * hoursEach;
        const laborCost = totalLaborHours * positive($("laborRate").value);
        const travelCost = positive($("miles").value) * positive($("mileCost").value);
        const materials = positive($("materialsCost").value);
        const other = positive($("otherCost").value);
        const overhead = positive($("overheadCost").value);
        const directCosts = laborCost + travelCost + materials + other + overhead;

        const processingRate = positive($("processingRate").value) / 100;
        const paymentFee = total * processingRate;
        const totalCosts = directCosts + paymentFee;
        const profit = serviceRevenue - totalCosts;
        const margin = serviceRevenue > 0 ? (profit / serviceRevenue) * 100 : 0;
        const effectiveHour = totalLaborHours > 0 ? profit / totalLaborHours : 0;
        const targetMargin = Math.min(90, positive($("targetMargin").value)) / 100;
        const denominator = Math.max(.05, 1 - targetMargin - processingRate * (1 + taxRate));
        const recommended = directCosts > 0 ? directCosts / denominator : 0;
        const increaseNeeded = Math.max(0, recommended - serviceRevenue);

        return {
          subtotal, discount, serviceRevenue, tax, total, deposit, balance,
          totalLaborHours, laborCost, travelCost, directCosts, paymentFee,
          totalCosts, profit, margin, effectiveHour, targetMargin: targetMargin * 100,
          recommended, increaseNeeded
        };
      }

      function previewDescription(item, language) {
        if (language === "en") return `<span>${escapeHtml(item.descriptionEn || "Service")}</span>`;
        if (language === "es") return `<span>${escapeHtml(item.descriptionEs || item.descriptionEn || "Servicio")}</span>`;
        const primary = item.descriptionEn || item.descriptionEs || "Service";
        const secondary = item.descriptionEs || item.descriptionEn || "";
        return `<span>${escapeHtml(primary)}</span>${secondary && secondary !== primary ? `<span class="item-secondary">${escapeHtml(secondary)}</span>` : ""}`;
      }

      function notesForLanguage(language) {
        const noteEn = $("notesEn").value.trim();
        const noteEs = $("notesEs").value.trim();
        const termsEn = $("termsEn").value.trim();
        const termsEs = $("termsEs").value.trim();
        if (language === "en") return [noteEn, termsEn].filter(Boolean).join("\n\n");
        if (language === "es") return [noteEs, termsEs].filter(Boolean).join("\n\n");
        return [
          noteEn && `EN: ${noteEn}`,
          termsEn && `EN: ${termsEn}`,
          noteEs && `ES: ${noteEs}`,
          termsEs && `ES: ${termsEs}`
        ].filter(Boolean).join("\n\n");
      }

      function calculateAndRender() {
        const calc = calculations();
        const language = $("quoteLanguage").value || "en";
        const type = documentType();
        const t = type === "invoice" ? invoiceText[language] : quoteText[language];
        if (!state.quoteNumber) state.quoteNumber = makeQuoteNumber();

        $("subtotalValue").textContent = currency(calc.subtotal);
        $("discountValue").textContent = currency(-calc.discount);
        $("taxValue").textContent = currency(calc.tax);
        $("grandTotalValue").textContent = currency(calc.total);

        $("profitValue").textContent = currency(calc.profit);
        $("marginValue").textContent = `${calc.margin.toFixed(1)}%`;
        $("jobCostsValue").textContent = currency(calc.totalCosts);
        $("effectiveHourValue").textContent = currency(calc.effectiveHour);
        $("recommendedValue").textContent = currency(calc.recommended);
        $("profitMetric").classList.toggle("negative", calc.profit < 0);

        const pill = $("healthPill");
        pill.className = "health-pill";
        if (calc.serviceRevenue <= 0) {
          pill.textContent = i18n[state.uiLanguage].waiting;
          $("recommendationText").textContent = i18n[state.uiLanguage].addCosts;
        } else if (calc.profit < 0) {
          pill.textContent = i18n[state.uiLanguage].loss;
          pill.classList.add("bad");
          $("recommendationText").textContent = i18n[state.uiLanguage].priceLoss;
        } else if (calc.margin + .01 < calc.targetMargin) {
          pill.textContent = i18n[state.uiLanguage].low;
          $("recommendationText").innerHTML = `${escapeHtml(i18n[state.uiLanguage].priceLow)} <strong>${currency(calc.increaseNeeded)}</strong>.`;
        } else {
          pill.textContent = i18n[state.uiLanguage].healthy;
          pill.classList.add("good");
          $("recommendationText").textContent = i18n[state.uiLanguage].priceOnTarget;
        }

        document.documentElement.style.setProperty("--accent", $("accentColor").value || "#087f73");
        document.documentElement.style.setProperty("--accent-dark", darkenHex($("accentColor").value || "#087f73", 18));
        document.documentElement.style.setProperty("--accent-soft", `${$("accentColor").value || "#087f73"}18`);

        setPreviewLabels(language);
        $("previewBusinessName").textContent = $("businessName").value.trim() || (state.uiLanguage === "es" ? "Nombre de tu negocio" : "Your Business Name");
        $("previewBusinessAddress").textContent = $("businessAddress").value.trim() || (state.uiLanguage === "es" ? "Ciudad, Estado" : "City, State");

        $("previewCustomerEmail").textContent = $("customerEmail").value.trim();

        $("previewQuoteNumber").textContent = state.quoteNumber;
        if (!$("documentDate").value) $("documentDate").value = isoToday();
